from collections import deque

def bfs(grafo, inicio):
    visitados = set()
    cola = deque([inicio])
    resultado = []
    while cola:
        nodo = cola.popleft()
        if nodo in visitados:
            continue
        visitados.add(nodo)
        resultado.append(nodo)
        for vecino in grafo.get(nodo, []):
            if vecino not in visitados:
                cola.append(vecino)

from functools import wraps

def memo(ttl=None):
    def decorator(func):
        cache = {}
        @wraps(func)
        def wrapper(*args):
            key = args
            now = time.time()
            if key in cache:
                result, ts = cache[key]
                if ttl is None or now - ts < ttl:
